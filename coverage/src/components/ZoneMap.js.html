    attachEvents(container) {
        const zones = container.querySelectorAll('.zone-group');
        zones.forEach(zone => {
            zone.addEventListener('mouseover', () => {
                const rect = zone.querySelector('.zone-rect');
                rect.style.fillOpacity = '0.4';
                rect.style.transform = 'translateY(-2px)';
            });
 
            zone.addEventListener('mouseout', () => {
                const zoneId = parseInt(zone.getAttribute('data-zone'));
                if (this.selectedZoneId !== zoneId) {
                    const rect = zone.querySelector('.zone-rect');
                    rect.style.fillOpacity = '0.2';
                    rect.style.transform = 'translateY(0)';
                }
            });
 
            zone.addEventListener('click', () => {
                const zoneId = parseInt(zone.getAttribute('data-zone'));
                this.selectZone(zoneId, container);
            });
        });
    }
 
    async selectZone(id, container) {
        this.selectedZoneId = id;
 
        // Reset other zones
        container.querySelectorAll('.zone-rect').forEach(rect => {
            rect.style.fillOpacity = '0.2';
            rect.style.strokeWidth = '2';
        });
 
        // Highlight selected
        const selectedRect = container.querySelector(`[data-zone="${id}"] .zone-rect`);
        selectedRect.style.fillOpacity = '0.5';
        selectedRect.style.stroke = 'var(--accent-neon)';
        selectedRect.style.strokeWidth = '4';
 
        // Update Detail Panel
        this.renderDetailPanel(id, container);
    }
