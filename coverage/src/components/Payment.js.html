export default class Payment {
  constructor(reservation, onNext, onPrev) {
    this.reservation = reservation;
    this.onNext = onNext;
    this.onPrev = onPrev;
    this.depositPercent = 30; // Default 30%
  }
 
  render(container) {
    const { zone, extraTable } = this.reservation;
    const totalMinSpend = zone.minSpend + (extraTable ? zone.extraTableCost : 0);
    const depositAmount = (totalMinSpend * this.depositPercent) / 100;
 
    container.innerHTML = `
      <div class="payment-container animate-fade">
        <button id="btn-back" class="btn btn-ghost" style="margin-bottom: var(--spacing-lg); padding: 8px 16px;">← ย้อนกลับ</button>
        
        <div style="display: grid; grid-template-columns: 1fr 350px; gap: var(--spacing-xl); align-items: start;">
          <div class="glass-card">
            <h2 style="margin-bottom: var(--spacing-md);">ชำระเงิน</h2>
            <p style="color: var(--text-muted); margin-bottom: var(--spacing-lg);">เลือกรูปแบบและช่องทางการชำระเงิน</p>
            
            <div style="margin-bottom: var(--spacing-xl);">
              <label style="display: block; margin-bottom: var(--spacing-md); font-weight: 500;">รูปแบบการชำระ</label>
              <div style="display: grid; grid-template-columns: 1fr 1fr; gap: var(--spacing-md);">
                <div class="payment-option ${this.depositPercent === 30 ? 'active' : ''}" data-percent="30">
                  <p style="font-size: 0.8rem; color: var(--text-muted);">มัดจำ 30%</p>
                  <p style="font-weight: bold;">฿${((totalMinSpend * 30) / 100).toLocaleString()}</p>
                </div>
                <div class="payment-option ${this.depositPercent === 100 ? 'active' : ''}" data-percent="100">
                  <p style="font-size: 0.8rem; color: var(--text-muted);">ชำระเต็มจำนวน</p>
                  <p style="font-weight: bold;">฿${totalMinSpend.toLocaleString()}</p>
                </div>
              </div>
            </div>
 
            <div>
              <label style="display: block; margin-bottom: var(--spacing-md); font-weight: 500;">ช่องทางการชำระ</label>
              <div style="display: flex; flex-direction: column; gap: var(--spacing-sm);">
                <label class="method-card">
                  <input type="radio" name="method" value="promptpay" checked>
                  <img src="https://upload.wikimedia.org/wikipedia/commons/c/c5/PromptPay_logo.png" alt="PromptPay" style="height: 20px;">
                  <span>Thai QR PromptPay</span>
                </label>
                <label class="method-card">

        .method-card input { width: 20px; height: 20px; accent-color: var(--primary); }

    this.attachEvents(container);

  attachEvents(container) {
    const options = container.querySelectorAll('.payment-option');
    const finalAmount = container.querySelector('#final-amount');
    const { zone, extraTable } = this.reservation;
    const totalMinSpend = zone.minSpend + (extraTable ? zone.extraTableCost : 0);
 
    options.forEach(opt => {
      opt.addEventListener('click', () => {
        options.forEach(o => o.classList.remove('active'));
        opt.classList.add('active');
        this.depositPercent = parseInt(opt.getAttribute('data-percent'));
        const amount = (totalMinSpend * this.depositPercent) / 100;
        finalAmount.textContent = `฿${amount.toLocaleString()}`;
      });
    });
 
    container.querySelector('#btn-back').addEventListener('click', () => {
      this.onPrev();
    });
 
    container.querySelector('#btn-submit').addEventListener('click', () => {
      // Collect selected method
      const method = container.querySelector('input[name="method"]:checked').value;
      this.reservation.payment = {
        method,
        depositPercent: this.depositPercent,
        depositAmount: (totalMinSpend * this.depositPercent) / 100,
        totalAmount: totalMinSpend
      };
      this.onNext();
    });
  }
}
